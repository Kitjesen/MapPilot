cmake_minimum_required(VERSION 3.20)
project(lingtu_online_mapping LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(LINGTU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
find_package(Threads REQUIRED)
if(NOT TARGET lingtu_lio_sam)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../sam" "${CMAKE_CURRENT_BINARY_DIR}/sam")
endif()
add_library(lingtu_online_mapping STATIC ../online_mapping.cpp ../cloud_io.cpp)
target_include_directories(lingtu_online_mapping PUBLIC "${LINGTU_ROOT}/src")
target_link_libraries(lingtu_online_mapping PUBLIC lingtu_lio_sam Threads::Threads)
add_executable(lt_mapping_replay "${LINGTU_ROOT}/tools/diagnostics/mapping_replay.cpp")
target_link_libraries(lt_mapping_replay PRIVATE lingtu_online_mapping)
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
  set(SAM_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(SAM_YAML_TARGET yaml-cpp)
endif()
target_link_libraries(lt_mapping_replay PRIVATE ${SAM_YAML_TARGET})
include(CTest)
if(BUILD_TESTING)
  add_executable(online_mapping_test "${LINGTU_ROOT}/tests/localization/cpp/opt/online_mapping_test.cpp")
  target_link_libraries(online_mapping_test PRIVATE lingtu_online_mapping ${SAM_YAML_TARGET})
  add_test(NAME online_mapping COMMAND online_mapping_test)
  set_tests_properties(online_mapping PROPERTIES TIMEOUT 60)
endif()

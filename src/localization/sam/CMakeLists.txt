cmake_minimum_required(VERSION 3.20)
project(lingtu_lio_sam LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(GTSAM 4.0 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters registration)
add_library(lingtu_lio_sam STATIC backend.cpp)
target_include_directories(lingtu_lio_sam PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${PCL_INCLUDE_DIRS})
target_link_libraries(lingtu_lio_sam PUBLIC gtsam ${PCL_LIBRARIES})
include(CTest)
if(BUILD_TESTING)
  add_executable(lio_sam_test ../../../tests/localization/cpp/sam/backend_test.cpp)
  target_link_libraries(lio_sam_test PRIVATE lingtu_lio_sam)
  add_test(NAME lio_sam_backend COMMAND lio_sam_test)
endif()

cmake_minimum_required(VERSION 3.20)
project(lingtu_online_mapping LANGUAGES C CXX)

include("${CMAKE_CURRENT_LIST_DIR}/../../../scripts/build/cmake/settings.cmake")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(LINGTU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/build/cmake/gtsam.cmake")
find_package(PCL REQUIRED COMPONENTS common filters registration)
add_library(lingtu_online_mapping_backend STATIC backend.cpp)
target_include_directories(lingtu_online_mapping_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${PCL_INCLUDE_DIRS})
target_link_libraries(lingtu_online_mapping_backend PUBLIC "$<BUILD_INTERFACE:lingtu_settings>" gtsam ${PCL_LIBRARIES})

find_package(Threads REQUIRED)
add_library(lingtu_online_mapping STATIC worker.cpp)
target_include_directories(lingtu_online_mapping PUBLIC "${LINGTU_ROOT}/src")
target_link_libraries(lingtu_online_mapping PUBLIC lingtu_online_mapping_backend Threads::Threads)

if(NOT TARGET lingtu_mapping_io)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../map_optimization/io"
    "${CMAKE_CURRENT_BINARY_DIR}/mapping_io")
endif()
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
  set(ONLINE_MAPPING_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(ONLINE_MAPPING_YAML_TARGET yaml-cpp)
endif()
add_executable(lt_mapping_replay "${LINGTU_ROOT}/tools/diagnostics/mapping_replay.cpp")
target_link_libraries(lt_mapping_replay PRIVATE
  lingtu_online_mapping lingtu_mapping_io ${ONLINE_MAPPING_YAML_TARGET})
if(UNIX AND NOT APPLE)
  include(GNUInstallDirs)
  foreach(_gtsam_runtime IN ITEMS gtsam metis-gtsam cephes-gtsam)
    if(TARGET ${_gtsam_runtime})
      get_target_property(_runtime_type ${_gtsam_runtime} TYPE)
      if(_runtime_type STREQUAL "SHARED_LIBRARY")
        install(FILES "$<TARGET_FILE:${_gtsam_runtime}>" "$<TARGET_SONAME_FILE:${_gtsam_runtime}>"
          DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT lingtu_runtime)
      endif()
    endif()
  endforeach()
  # The prepared SDK keeps upstream notices alongside the binary dependency.
  install(DIRECTORY "${GTSAM_INCLUDE_DIR}/../share/lingtu/gtsam-licenses"
    DESTINATION ${CMAKE_INSTALL_DATADIR}/lingtu COMPONENT lingtu_runtime OPTIONAL)
endif()

  add_executable(online_mapping_confirmation_test ../../../tests/localization/cpp/online_mapping/confirmation_test.cpp)
  target_link_libraries(online_mapping_confirmation_test PRIVATE lingtu_online_mapping_backend)
  add_test(NAME online_mapping_confirmation COMMAND online_mapping_confirmation_test)

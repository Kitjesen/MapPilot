cmake_minimum_required(VERSION 3.20)
project(lingtu_online_mapping_backend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../scripts/build/cmake/gtsam.cmake")
find_package(PCL REQUIRED COMPONENTS common filters registration)
add_library(lingtu_online_mapping_backend STATIC backend.cpp)
target_include_directories(lingtu_online_mapping_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${PCL_INCLUDE_DIRS})
target_link_libraries(lingtu_online_mapping_backend PUBLIC gtsam ${PCL_LIBRARIES})
if(UNIX AND NOT APPLE)
  include(GNUInstallDirs)
  foreach(_gtsam_runtime IN ITEMS gtsam metis-gtsam cephes-gtsam)
    if(TARGET ${_gtsam_runtime})
      get_target_property(_runtime_type ${_gtsam_runtime} TYPE)
      if(_runtime_type STREQUAL "SHARED_LIBRARY")
        install(FILES "$<TARGET_FILE:${_gtsam_runtime}>" "$<TARGET_SONAME_FILE:${_gtsam_runtime}>"
          DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT lingtu_runtime)
      endif()
    endif()
  endforeach()
  # The prepared SDK keeps upstream notices alongside the binary dependency.
  install(DIRECTORY "${GTSAM_INCLUDE_DIR}/../share/lingtu/gtsam-licenses"
    DESTINATION ${CMAKE_INSTALL_DATADIR}/lingtu COMPONENT lingtu_runtime OPTIONAL)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(online_mapping_backend_test ../../../tests/localization/cpp/online_mapping/backend_test.cpp)
  target_link_libraries(online_mapping_backend_test PRIVATE lingtu_online_mapping_backend)
  add_test(NAME online_mapping_backend COMMAND online_mapping_backend_test)
  add_executable(online_mapping_confirmation_test ../../../tests/localization/cpp/online_mapping/confirmation_test.cpp)
  target_link_libraries(online_mapping_confirmation_test PRIVATE lingtu_online_mapping_backend)
  add_test(NAME online_mapping_confirmation COMMAND online_mapping_confirmation_test)
  add_executable(online_mapping_candidates_test ../../../tests/localization/cpp/online_mapping/candidates_test.cpp)
  target_link_libraries(online_mapping_candidates_test PRIVATE lingtu_online_mapping_backend)
  add_test(NAME online_mapping_candidates COMMAND online_mapping_candidates_test)
  set_tests_properties(online_mapping_candidates PROPERTIES TIMEOUT 60)
  add_executable(online_mapping_risam_test ../../../tests/localization/cpp/online_mapping/risam_test.cpp)
  target_link_libraries(online_mapping_risam_test PRIVATE lingtu_online_mapping_backend)
  add_test(NAME online_mapping_risam COMMAND online_mapping_risam_test)
  set_tests_properties(online_mapping_risam PROPERTIES TIMEOUT 60)
endif()

cmake_minimum_required(VERSION 3.16)
project(lingtu_erasor2_research LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

get_filename_component(LINGTU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ERASOR2_SOURCE_DIR "${LINGTU_ROOT}/third_party/research_nav/ERASOR2" CACHE PATH "Pinned upstream ERASOR2 checkout")
option(LINGTU_ERASOR2_USE_RERUN_STUB "Use the no-op research visualization stub" ON)
if(LINGTU_ERASOR2_USE_RERUN_STUB)
  set(FETCHCONTENT_SOURCE_DIR_RERUN_SDK "${CMAKE_CURRENT_SOURCE_DIR}/rerun_stub" CACHE PATH "No-op rerun SDK" FORCE)
endif()
add_subdirectory("${ERASOR2_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/erasor2_upstream")
set_target_properties(run_erasor2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
add_library(erasor2_stage_core STATIC stager.cpp)
target_include_directories(erasor2_stage_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(erasor2_stage stage.cpp)
target_link_libraries(erasor2_stage PRIVATE erasor2_stage_core)
add_executable(erasor2_clean clean.cpp)
add_dependencies(erasor2_clean erasor2_stage run_erasor2)
if(BUILD_TESTING)
  add_test(NAME erasor2_stage_help COMMAND erasor2_stage --help)
  add_test(NAME erasor2_clean_help COMMAND erasor2_clean --help)
endif()

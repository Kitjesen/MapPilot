cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(lingtu_maps_ground LANGUAGES CXX)
  include(CTest)
endif()

if(NOT TARGET lingtu_maps_ground)
  find_package(Eigen3 3.3 REQUIRED NO_MODULE)
  add_library(lingtu_maps_ground STATIC segmenter.cpp vendor/patchworkpp.cpp vendor/plane_fit.cpp)
  target_compile_features(lingtu_maps_ground PUBLIC cxx_std_20)
  target_include_directories(lingtu_maps_ground PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
  target_include_directories(lingtu_maps_ground PRIVATE vendor)
  target_link_libraries(lingtu_maps_ground PRIVATE Eigen3::Eigen)
  set_target_properties(lingtu_maps_ground PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(BUILD_TESTING)
    add_executable(ground_segmenter_test "${CMAKE_CURRENT_SOURCE_DIR}/../tests/ground_segmenter_test.cpp")
    target_include_directories(ground_segmenter_test PRIVATE vendor)
    target_link_libraries(ground_segmenter_test PRIVATE lingtu_maps_ground Eigen3::Eigen)
    foreach(test_case narrow_continuity height_continuity reset cold_inputs nonfinite)
      add_test(NAME ground_${test_case} COMMAND ground_segmenter_test ${test_case})
    endforeach()
  endif()
endif()
